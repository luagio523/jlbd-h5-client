export class ModifierEventDispatcher {

    protected eventListeners: Map<ModifierEvent, Set<(event: ModifierEvent, params: any) => void>> = new Map();

    public addEventListener(eventType: ModifierEvent, listener: (event: ModifierEvent, params: any) => void): void {
        if (!this.eventListeners.has(eventType)) {
            this.eventListeners.set(eventType, new Set());
        }
        this.eventListeners.get(eventType)!.add(listener);
    }

    public removeEventListener(eventType: ModifierEvent, listener: (event: ModifierEvent, params: any) => void): void {
        if (this.eventListeners.has(eventType)) {
            this.eventListeners.get(eventType)!.delete(listener);
        }
    }

    public dispatchEvent(event: ModifierEvent, params: any): void {
        if (this.eventListeners.has(event)) {
            this.eventListeners.get(event)!.forEach((listener) => listener(event,params));
        }
    }

    private constructor() {}

    public static getInstance(): ModifierEventDispatcher {
        if (!ModifierEventDispatcher.instance) {
            ModifierEventDispatcher.instance = new ModifierEventDispatcher();
        }
        return ModifierEventDispatcher.instance;
    }

    private static instance: ModifierEventDispatcher;
   
}




export abstract class Modifier {
    public stackCount: number = 1;
    public attribute: ModifierAttribute;

    protected caster: ECSEntity | null = null;
    protected ability: BaseAbility | null = null;
    protected auraOwner: ECSEntity | null = null;
    protected timer: Timer;

    constructor(
        public id: string,
        public duration: number = Infinity,
        public isHidden: boolean = false,
        public isDebuff: boolean = false,
        public purgable: boolean = true,
        public priority: number = 0,
        attribute: ModifierAttribute = ModifierAttribute.MODIFIER_ATTRIBUTE_NONE
    ) {
        this.attribute = attribute;
    }

    public update(deltaTime: number): void {
        if (this.timer) {
            this.timer.update(deltaTime);
        }
        if (this.duration !== Infinity) {
            this.duration -= deltaTime;
            if (this.duration <= 0) {
                this.OnDestroy();
            }
        }
    }

   
    public IsDebuff(): boolean { return this.isDebuff; }
    public IsPurgable(): boolean { return this.purgable; }

    public OnCreated(): void {}
    public OnRefresh(): void {}
    public OnDestroy(): void {}

    protected SetStackCount(count: number): void { this.stackCount = count; }
    protected IncrementStackCount(): void { this.stackCount++; }

    protected StartIntervalThink(thinkInterval: number) {
        if (!this.timer) {
            this.timer = new Timer(thinkInterval, this.OnIntervalThink.bind(this));
        }
        this.timer.setInterval(thinkInterval);
    }

    protected OnIntervalThink() {}

    // abstract ListModifierCategory(): ModifierCategory[]; // 作用范围

    abstract ListModifierFunctions(): ModifierFunction[]; // 作用属性


    abstract ListListenEvents(): ModifierEvent[];

    getModifierValue(func: ModifierFunction): number {
        const declaredFunctions = this.ListModifierFunctions();
        if (declaredFunctions.includes(func)) {
            return this.getModifierValueInternal(func);
        }
        return 0;
    }

    onModifierEvent(event: ModifierEvent, params: any): void{
        this.onModifierEventInternal(event, params);
    };

    protected abstract onModifierEventInternal(event: ModifierEvent, params: any): void;

    protected abstract getModifierValueInternal(func: ModifierFunction): number;

    abstract checkState(): Map<ModifierState, boolean>;

    getModifierState(state: ModifierState): boolean {
        return this.checkState().get(state) || false;
    }

    public handleStacking(newModifier: Modifier) {
        if (this.hasAttribute(ModifierAttribute.MODIFIER_ATTRIBUTE_MULTIPLE)) {
            this.IncrementStackCount();
        } else if (this.hasAttribute(ModifierAttribute.MODIFIER_ATTRIBUTE_STACKABLE)) {
            this.stackCount += newModifier.stackCount;
        } else if (this.hasAttribute(ModifierAttribute.MODIFIER_ATTRIBUTE_REFRESH_ON_STACK)) {
            this.duration = newModifier.duration;
        } else if (this.hasAttribute(ModifierAttribute.MODIFIER_ATTRIBUTE_UNIQUE)) {
        // 不做任何操作，保持唯一性
        } else {
        // 默认行为：刷新持续时间
            this.duration = newModifier.duration;
        }
        this.OnRefresh();
    }

    

    public setCaster(caster: ECSEntity): void { this.caster = caster; }
    public getCaster(): ECSEntity | null { return this.caster; }
    public setAbility(ability: BaseAbility): void { this.ability = ability; }
    public getAbility(): BaseAbility | null { return this.ability; }
    public setAuraOwner(owner: ECSEntity): void { this.auraOwner = owner; }
    public getAuraOwner(): ECSEntity | null { return this.auraOwner; }

    public isAura(): boolean { return false; }
    public createAuraEffect(target: ECSEntity): Modifier | null { return null; }
    public getAuraRadius(): number { return 0; }
    // public shouldApplyToTarget(target: ECSEntity): boolean { return true; }

    public hasAttribute(attr: ModifierAttribute): boolean {
        return (this.attribute & attr) !== 0;
    }

    public getRemainingDuration(): number { return this.duration; }
    public isExpired(): boolean { return this.duration <= 0; }

    public abstract getAffectedComponentTypes(): Set<Function>;
    public abstract clone(): Modifier;


    public shouldRemoveOnDeath(): boolean {
        return this.hasAttribute(ModifierAttribute.MODIFIER_ATTRIBUTE_REMOVE_ON_DEATH);
    }

    public IsHidden(): boolean {
        return this.hasAttribute(ModifierAttribute.MODIFIER_ATTRIBUTE_HIDDEN);
    }

    public ignoreTenacity(): boolean {
    return this.hasAttribute(ModifierAttribute.MODIFIER_ATTRIBUTE_IGNORE_TENACITY);
    }
}
export class ModifierComponent extends ECSComponent {

    private modifierEventDispatcher: ModifierEventDispatcher = ModifierEventDispatcher.getInstance();

    private modifiers: Modifier[] = [];
    private dirty: boolean = true;
    private cache: Map<ModifierFunction, number> = new Map();

    onModifierEvent(event: ModifierEvent, data: any) {
        this.modifierEventDispatcher.dispatchEvent(event, data);
    }

    addModifier(modifier: Modifier) {
        modifier.ListListenEvents().forEach(event => {
            this.modifierEventDispatcher.addEventListener(event, modifier.onModifierEvent);
        });
        this.modifiers.push(modifier);
        // 处理光环优先级
        if (modifier.hasAttribute(ModifierAttribute.MODIFIER_ATTRIBUTE_AURA_PRIORITY)) {
            this.modifiers.sort((a, b) => {
                const aHasAuraPriority = a.hasAttribute(ModifierAttribute.MODIFIER_ATTRIBUTE_AURA_PRIORITY);
                const bHasAuraPriority = b.hasAttribute(ModifierAttribute.MODIFIER_ATTRIBUTE_AURA_PRIORITY);

                if (aHasAuraPriority && !bHasAuraPriority) {
                    return -1;
                }
                if (!aHasAuraPriority && bHasAuraPriority) {
                    return 1;
                }
                return b.priority - a.priority;
            });
        } else {
            this.modifiers.sort((a, b) => b.priority - a.priority);
        }
        this.markDirty();
    }


    removeModifier(modifierId: string) {
        const index = this.modifiers.findIndex(m => m.id === modifierId);
        const modifier = this.modifiers[index];
        if (modifier) {
            modifier.ListListenEvents()?.forEach(event => {
                this.modifierEventDispatcher.removeEventListener(event,modifier.onModifierEvent);
            });
        }
        if (index !== -1) {
            this.modifiers.splice(index, 1);
            this.markDirty();
        }
    }

    getModifiers(): Modifier[] {
        return this.modifiers;
    }

    markDirty() {
        this.dirty = true;
        this.cache.clear();
    }

    isDirty(): boolean {
        return this.dirty;
    }


    clearDirty() {
        this.dirty = false;
    }

    private calculateModifierValue( func: ModifierFunction): number {
        return this.getModifiers().reduce((total, modifier) => {
            return total + modifier.getModifierValue(func);
        }, 0);
    }


    private recalculateCache(): void {
        for (const modifier of this.getModifiers()) {
            for (const func of modifier.ListModifierFunctions()) {
                const value = this.calculateModifierValue(func);
                this.cache.set(func, value);
            }
        }
        this.clearDirty();
    }


    public getModifierValue(func: ModifierFunction): number {
        if (this.isDirty()) {
            this.recalculateCache();
        }
        return this.cache.get(func) || 0;
    }


    public isInState(state: ModifierState): boolean {
        return this.getModifiers().some(modifier => modifier.getModifierState(state));
    }
}



export class AttackSpeedCalculator {
    private modifierManager: ModifierManager;
    private attributeCalculator: AttributeCalculator;
    constructor(modifierManager: ModifierManager) {
        this.modifierManager = modifierManager;
        this.attributeCalculator = new AttributeCalculator(this.modifierManager);
    }

    // 计算最终攻击速度
    calculateFinalAttackRate(entity: ECSEntity, baseAttackTime: number): number {
        // 计算基础攻击时间 (BAT)
        let finalBAT = this.calculateBAT(entity, baseAttackTime);

        // 计算攻击速度
        let attackSpeedBonus = this.calculateFinalAttackSpeedBonus(entity,0); //默认基础速度是0

        let multiplier = this.calculateAttackIntervalMultiplier(attackSpeedBonus);
        // 计算攻击时间
        let attackTime = finalBAT / multiplier;
        // 应用攻击速度限制
        let ignoreLimit = this.modifierManager.getModifierValue(entity, ModifierFunction.MODIFIER_PROPERTY_IGNORE_ATTACKSPEED_LIMIT);
        if (!ignoreLimit) {
            attackTime = Math.max(Constants.MIN_ATTACK_TIME, Math.min(Constants.MAX_ATTACK_TIME, attackTime));
        }
        // 将攻击时间转换为每秒攻击次数
        return attackTime;
    }

    calculateAttackIntervalMultiplier(attackSpeedBonus: number): number {
        return Math.pow(2, attackSpeedBonus / 100);
    }

    // 计算基础攻击时间 (BAT)
    private calculateBAT(entity: ECSEntity, baseAttackTime: number): number {
        let bat = baseAttackTime;

        // // 应用 BAT 常数修改器
        // let batConstant = this.modifierManager.getModifierValue(entity, ModifierFunction.MODIFIER_PROPERTY_BASE_ATTACK_TIME_CONSTANT);
        // if (batConstant > 0) {
        //     bat = batConstant;
        // }

        // // 应用 BAT 调整
        // let batAdjust = this.modifierManager.getModifierValue(entity, ModifierFunction.MODIFIER_PROPERTY_BASE_ATTACK_TIME_CONSTANT_ADJUST);
        // bat += batAdjust;

        // // 应用 BAT 百分比修改器
        // let batPercentage = this.modifierManager.getModifierValue(entity, ModifierFunction.MODIFIER_PROPERTY_BASE_ATTACK_TIME_PERCENTAGE);
        // bat *= (1 + batPercentage / 100);

        return bat;
    }

    // 计算攻击速度
    public calculateFinalAttackSpeedBonus(entity: ECSEntity,baseAttackSpeed: number): number {
        const agility = entity.getComponent(AttributeComponent).getAgility();
        const attributeAttackSpped = this.attributeCalculator.calculateFinalAgility(entity,agility)*1 // 每点敏捷提供1点攻击速度
        baseAttackSpeed += attributeAttackSpped // 加上属性加成
        // 应用常数加成
        baseAttackSpeed += this.modifierManager.getModifierValue(entity, ModifierFunction.MODIFIER_PROPERTY_ATTACKSPEED_BONUS_CONSTANT);
        return baseAttackSpeed;
    }

    
}

export class AttackComponent extends ECSComponent {
    public attackCapability: AttackCapability;
    public baseAttackDamageMin: number;
    public baseAttackDamageMax: number;
    public attackRange: number;
    public attackRate: number;
    public attackAnimationPoint: number;
    public attackAcquisitionRange: number;
    public attackType: AttackType;
    public attackDamageType: DamageType;
    private target: ECSEntity | null = null;
    private attacksMade: number = 0;
    private attackCooldown: Cooldown;

    constructor(heroData: HeroData) {
        super();
        this.initializeAttack(heroData);
    }
   

    private initializeAttack(heroData: HeroData): void {
        this.attackCapability = this.determineAttackCapability(heroData.GetAttackRange());
        [this.baseAttackDamageMin, this.baseAttackDamageMax] = heroData.GetAttackDamage();
        this.attackRange = heroData.GetAttackRange();
        this.attackRate = heroData.GetAttackRate();
        this.attackAnimationPoint = heroData.GetAttackAnimationPoint();
        this.attackAcquisitionRange = heroData.GetAttackAcquisitionRange();
        this.attackType = heroData.GetAttackType();
        this.attackDamageType = heroData.GetAttackDamageType();
        this.attackCooldown = new Cooldown(this.attackRate);
    }

    private determineAttackCapability(attackRange: number): AttackCapability {
        return attackRange > 150 ? AttackCapability.UNIT_CAP_RANGED_ATTACK : AttackCapability.UNIT_CAP_MELEE_ATTACK;
    }

    public getBaseAttackDamage(): number {
        return Math.random() * (this.baseAttackDamageMax - this.baseAttackDamageMin) + this.baseAttackDamageMin;
    }

    public getAttackRange(): number {
        return this.attackRange;
    }

    

    public setTarget(target: ECSEntity | null): void {
        this.target = target;
    }

    public getTarget(): ECSEntity | null {
        return this.target;
    }

    public canAttack(deltaTime: number): boolean {
        return this.attackCooldown.check(deltaTime);
    }

    public updateAttackRate(attackRate: number): void {
        this.attackRate = attackRate;
        this.attackCooldown.resetCooldownRate(attackRate);
    }

    public performAttack(): void {
        if (this.canAttack(0)) {
            this.attacksMade++;
            this.attackCooldown.start();
        }
    }

    public getAttacksMade(): number {
        return this.attacksMade;
    }

    public getAttackAnimationPoint(): number {
        return this.attackAnimationPoint;
    }

    public getAttackAcquisitionRange(): number {
        return this.attackAcquisitionRange;
    }

    public getAttackType(): AttackType {
        return this.attackType;
    }

    public getAttackDamageType(): DamageType {
        return this.attackDamageType;
    }

    

    public resetAttacksMade(): void {
        this.attacksMade = 0;
    }

    public isInAcquisitionRange(distance: number): boolean {
        return distance <= this.attackAcquisitionRange;
    }

    public isInAttackRange(distance: number): boolean {
        return distance <= this.attackRange;
    }
}

export class AttackSystem extends BaseSystem {
    private readonly TRACK_INTERVAL: number = 0.5; // 追踪更新间隔
    private lastTrackTime: number = 0;

    update(deltaTime: number) {
        const currentTime = Date.now() / 1000;
        const attackComponents = this.getComponents(AttackComponent);

        for (const attackComponent of attackComponents) {
            const attacker = this.getEntity(attackComponent.entityId);
            const target = attackComponent.getTarget();

            if (target) {
                //是否每次都更新新的攻击间隔
                const distance = this.getDistanceBetweenEntities(attacker, target);

                if (attackComponent.isInAttackRange(distance)) {
                    if (attackComponent.canAttack(deltaTime)) {
                        this.initiateAttack(attacker, target, attackComponent);
                    }
                } else if (attackComponent.isInAcquisitionRange(distance)) {
                    if (currentTime - this.lastTrackTime >= this.TRACK_INTERVAL) {
                        this.updateTracking(attacker, target);
                        this.lastTrackTime = currentTime;
                    }
                } else {
                    // 目标超出获取范围，可以选择停止追踪或继续追踪
                    attackComponent.setTarget(null);
                }
            } else {
                // 可以在这里添加寻找新目标的逻辑
            }
        }
    }

    private initiateAttack(attacker: ECSEntity, target: ECSEntity, attackComponent: AttackComponent) {
        // 触发攻击开始事件
        this.enqueueEvent(new AttackEvent(attacker, target));

        // 触发动画事件
        this.enqueueEvent(new AnimationEvent(attacker, ActType.ACT_ATTACK));

        // 根据攻击类型决定是立即造成伤害还是延迟造成伤害
        if (attackComponent.attackCapability === AttackCapability.UNIT_CAP_MELEE_ATTACK) {
            this.enqueueEvent(new DelayedEvent(
                () => this.performAttack(attacker, target, attackComponent),
                attackComponent.getAttackAnimationPoint()
            ));
        } else if (attackComponent.attackCapability === AttackCapability.UNIT_CAP_RANGED_ATTACK) {
            // 触发投射物事件
            this.enqueueEvent(new ProjectileEvent(attacker, target));
        }

        // 执行攻击
        attackComponent.performAttack();
    }

    private performAttack(attacker: ECSEntity, target: ECSEntity, attackComponent: AttackComponent) {
        const damage = attackComponent.getBaseAttackDamage();
        const damageType = attackComponent.getAttackDamageType();

        this.enqueueEvent(new ApplyDamageEvent(attacker, target, damage, damageType));
    }

    private updateTracking(attacker: ECSEntity, target: ECSEntity) {
        const attackerTransform = attacker.getComponent(TransformComponent);
        const targetTransform = target.getComponent(TransformComponent);

        if (attackerTransform && targetTransform) {
            attackerTransform.setTargetPosition(targetTransform.position);

            // 更新寻路组件（如果有的话）
            const pathfinding = attacker.getComponent(PathfindingComponent);
            if (pathfinding) {
                pathfinding.setTarget(targetTransform.position);
            }
        }
    }

    private getDistanceBetweenEntities(entity1: ECSEntity, entity2: ECSEntity): number {
        const transform1 = entity1.getComponent(TransformComponent);
        const transform2 = entity2.getComponent(TransformComponent);

        if (transform1 && transform2) {
            return Vec2.distance(transform1.position, transform2.position);
        }

        return Infinity; // 如果无法计算距离，返回无穷大
    }
}
